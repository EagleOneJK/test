
function Delete-ELKPointInTime{
    param(
        [CmdletBinding()]
        [Parameter(Mandatory=$true)] [string]$PIT_id,
        [Parameter(Mandatory=$false)] [string]$ELKrestURI,
        [Parameter(Mandatory=$false)] [PSCredential]$ElkCredential,
        [Parameter(Mandatory=$false)] [switch]$ELKCredentialIsAPIKey
    )

    if(!$ElkCredential){
        $ElkCredential = Get-Credential -Message "Provide ELK Credential" -UserName $env:USERNAME
        $prompt = Read-Host -Prompt "Is this credential an API Key (y/N)?"
        if($prompt -match "^y"){$ElkCredentialIsAPIKey = $True}
    }

    # Convert creds to Base64
    $unsecureCreds = $ElkCredential.GetNetworkCredential()
    $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $unsecureCreds.UserName,$unsecureCreds.Password)))
    Remove-Variable unsecureCreds

    if($ElkCredentialIsAPIKey){
        $headers = @{Authorization=("ApiKey {0}" -f $base64AuthInfo)}
    }else{
        $headers = @{Authorization=("Basic {0}" -f $base64AuthInfo)}
    }

    $data = @{"id"=$PIT_id}

    $result = Invoke-RestMethod "$ELKrestURI/_pit" -Method Delete -Headers $headers -ContentType 'application/json' -Body ($data | ConvertTo-Json -Depth 15) #-UseBasicParsing
    $result
}

function Get-HashtableRecursiveContains{
    param([hashtable]$hashtable,[string[]]$NestedElements)
    #Write-Host -Debug -ForegroundColor Cyan $elements
    #Write-Host -Debug -ForegroundColor DarkCyan $obj
    if($hashtable -eq $null -or $NestedElements.Count -le 0){return $false}
    if($NestedElements.Count -eq 1){
        return ([hashtable]$hashtable).Contains($NestedElements[0])
    }else{
        [hashtable]$newHT = [hashtable]($hashtable.$($NestedElements[0]))
        $null,[object[]]$NewElements=$NestedElements
        return Get-HashtableRecursiveContains -hashtable ([hashtable]$newHT) -NestedElements $NewElements
    }
}

function ConvertTo-Hashtable {
    #https://4sysops.com/archives/convert-json-to-a-powershell-hash-table/
    [CmdletBinding()]
    [OutputType('hashtable')]
    param (
        [Parameter(ValueFromPipeline)]
        $InputObject
    )
    process {
        ## Return null if the input is null. This can happen when calling the function
        ## recursively and a property is null
        if ($null -eq $InputObject) {
            return $null
        }
        ## Check if the input is an array or collection. If so, we also need to convert
        ## those types into hash tables as well. This function will convert all child
        ## objects into hash tables (if applicable)
        if ($InputObject -is [System.Collections.IEnumerable] -and $InputObject -isnot [string]) {
            $collection = @(
                foreach ($object in $InputObject) {
                    ConvertTo-Hashtable -InputObject $object
                }
            )
            ## Return the array but don't enumerate it because the object may be pretty complex
            Write-Output -NoEnumerate $collection
        } elseif ($InputObject -is [psobject]) { ## If the object has properties that need enumeration
            ## Convert it to its own hash table and return it
            $hash = @{}
            foreach ($property in $InputObject.PSObject.Properties) {
                $hash[$property.Name] = ConvertTo-Hashtable -InputObject $property.Value
            }
            $hash
        } else {
            ## If the object isn't an array, collection, or other object, it's already a hash table
            ## So just return it.
            $InputObject
        }
    }
}

        [Parameter(Mandatory=$false)] [int]$SubBucketMinDocCount=1,
        [Parameter(Mandatory=$false)] [ValidateRange(0.1,99.9)] [double[]]$SubBucketPercents=50.0,

        [Parameter(Mandatory=$false)] [ValidateSet("terms","cardinality","sum","avg","percentiles","date_histogram")] [string[]]$SubSubBucketType="terms",
        [Parameter(Mandatory=$false)] [switch]$SubSubBucketDescriptiveNames,
        [Parameter(Mandatory=$false)] [string]$SubSubBucketField,
        [Parameter(Mandatory=$false)] [string]$SubSubBucketSortField,
        [Parameter(Mandatory=$false)] [ValidateSet("asc","desc")] [string]$SubSubBucketSortOrder="desc",
        [Parameter(Mandatory=$false)] [int]$SubSubBucketSize=3,
        [Parameter(Mandatory=$false)] [string]$SubSubBucketInterval="6h",
        [Parameter(Mandatory=$false)] [int]$SubSubBucketMinDocCount=1,
        [Parameter(Mandatory=$false)] [ValidateRange(0.1,99.9)] [double[]]$SubSubBucketPercents=50.0,

        [Parameter(Mandatory=$false)] [ValidateSet("terms","cardinality","sum","avg","percentiles","date_histogram")] [string[]]$SubSubSubBucketType="terms",
        [Parameter(Mandatory=$false)] [switch]$SubSubSubBucketDescriptiveNames,
        [Parameter(Mandatory=$false)] [string]$SubSubSubBucketField,
        [Parameter(Mandatory=$false)] [string]$SubSubSubBucketSortField,
        [Parameter(Mandatory=$false)] [ValidateSet("asc","desc")] [string]$SubSubSubBucketSortOrder="desc",
        [Parameter(Mandatory=$false)] [int]$SubSubSubBucketSize=3,
        [Parameter(Mandatory=$false)] [string]$SubSubSubBucketInterval="6h",
        [Parameter(Mandatory=$false)] [int]$SubSubSubBucketMinDocCount=1,
        [Parameter(Mandatory=$false)] [ValidateRange(0.1,99.9)] [double[]]$SubSubSubBucketPercents=50.0,

        [Parameter(Mandatory=$false)] [switch]$Scroll,
        [Parameter(Mandatory=$false)] [string]$ScrollID="",
        [Parameter(Mandatory=$false)] [int]$KeepAliveMinutes=15,
        [Parameter(Mandatory=$false)] [switch]$PointInTime,
        [Parameter(Mandatory=$false)] [string]$PointInTimeID="",
        [Parameter(Mandatory=$false)] [string[]]$PointInTimeSearchAfter="", #should be the value of the last "Sort" item in the previous list of results 
        [Parameter(Mandatory=$false)] [switch]$DebugQuery,
        [Parameter(Mandatory=$false)] [switch]$DebugRequest,
        [Parameter(Mandatory=$false)] [int]$TimeoutSec=0,
        [Parameter(Mandatory=$false)] [switch]$Profiler,
        [Parameter(Mandatory=$false)] [bool]$AnalyzeWildcards=$True
    )
    
    if(!$ElkCredential){
        $ElkCredential = Get-Credential -Message "Provide ELK Credential" -UserName $env:USERNAME
        $prompt = Read-Host -Prompt "Is this credential an API Key (y/N)?"
        if($prompt -match "^y"){$ElkCredentialIsAPIKey = $True}
    }

    if($ELKIndex -match "^hunt"){$TimeField = "event_time"}

    if($UseLocalTimeZone){[string]$TimeZone=[string]$TimeZone="{0:D2}:00" -f ([System.DateTimeOffset]::Now).Offset.Hours}

    if($AggregationJson -and $BucketField){
        Write-Host -Debug -ForegroundColor Red "Cannot supply JSON aggregation string and build dynamic aggregation query. Pick one."
        return
    }

    # Convert creds to Base64
    $unsecureCreds = $ElkCredential.GetNetworkCredential()
    $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $unsecureCreds.UserName,$unsecureCreds.Password)))
    Remove-Variable unsecureCreds

    if($ElkCredentialIsAPIKey){
        $headers = @{Authorization=("ApiKey {0}" -f $base64AuthInfo)} #$headers = @{Authorization=("ApiKey {0}" -f $base64AuthInfo); charset="UTF-8"}
    }else{
        $headers = @{Authorization=("Basic {0}" -f $base64AuthInfo)} #$headers = @{Authorization=("Basic {0}" -f $base64AuthInfo); charset="UTF-8"}
    }
    
    $start = ""
    $end=""

    if($startTime -or $endTime){
        if(!$endTime){$endTime = Get-Date}
        if(!$startTime){$startTime = (Get-Date).AddYears(-1)}
        if($startTime -ge $endTime){
            Write-Host -Debug -ForegroundColor Red "ERROR! Query Start Time must be before Query End time chronologically"
            return
        }
        [string]$start = $startTime.ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
        [string]$end = $endTime.ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
    }elseif($numHoursAgoStart -or $numHoursAgoEnd){
        if($numHoursAgoStart -le $numHoursAgoEnd){
            Write-Host -Debug -ForegroundColor Red "ERROR! Query Start Time must be before Query End time chronologically. numHoursAgoStart must be greater than numHoursAgoEnd."
            return
        }
        if($numHoursAgoStart -lt 0 -or $numHoursAgoEnd -lt 0){
            [string]$start = (Get-Date).AddHours(0-$numHoursAgoStart).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
            [string]$end = (Get-Date).AddHours(0-$numHoursAgoEnd).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
        }else{
            $interval = "h"
            [string]$start = "now-$($numHoursAgoStart)$interval"
            [string]$end = "now-$($numHoursAgoEnd)$interval"
        }
    }elseif($numDaysAgoStart -or $numDaysAgoEnd){
        if($numDaysAgoStart -le $numDaysAgoEnd){
            Write-Host -Debug -ForegroundColor Red "ERROR! Query Start Time must be before Query End time chronologically. numDaysAgoStart must be greater than numDaysAgoEnd."
            return
        }
        if($numDaysAgoStart -lt 0 -or $numDaysAgoEnd -lt 0){
            [string]$start = (Get-Date).AddDays(0-$numDaysAgoStart).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
            [string]$end = (Get-Date).AddDays(0-$numDaysAgoEnd).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
        }else{
            $interval = "d"
            [string]$start = "now-$($numDaysAgoStart)$interval"
            [string]$end = "now-$($numDaysAgoEnd)$interval"
        }
    }else{
        if($numMinutesAgoStart -le $numMinutesAgoEnd){
            Write-Host -Debug -ForegroundColor Red "ERROR! Query Start Time must be before Query End time chronologically. numMinutesAgoStart must be greater than numMinutesAgoEnd."
            return
        }
        if($numMinutesAgoStart -lt 0 -or $numMinutesAgoEnd -lt 0){
            [string]$start = (Get-Date).AddMinutes(0-$numMinutesAgoStart).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
            [string]$end = (Get-Date).AddMinutes(0-$numMinutesAgoEnd).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ss.ffffffZ")
        }else{
            $interval = "m"
            [string]$start = "now-$($numMinutesAgoStart)$interval"
            [string]$end = "now-$($numMinutesAgoEnd)$interval"
        }
    }

    $query = @{size=$RecordCount; query = @{bool = @{must = @(@{query_string = @{query = $ELKQuery; analyze_wildcard=$AnalyzeWildcards}});filter = @{range = @{$TimeField = @{gte = $start; lt = $end}}}}};sort = @{$SortField = @{order = $SortOrder}};}
    if($NoTimeField){
        $query = @{size=$RecordCount; query = @{bool = @{must = @(@{query_string = @{query = $ELKQuery; analyze_wildcard=$AnalyzeWildcards}})}}}
    }
    if($ExcludeSource){
        $query.Add('_source',$false)
    }
    if($Profiler){
        $query.Add('profile',$true)
    }
    if($Fields){
        if(!$ExcludeSource){$query.Add('_source',$false)}
        if(!$query.ContainsKey('fields')){$query.Add('fields',$Fields)
        }else{$query['fields'] += $Fields}
    }

    if($BucketField){
        $bucket_count = $BucketType.Count
        $this_bucket_number = 1
        foreach($bt in $BucketType){
            if(!$query.ContainsKey("aggs")){$query.aggs = @{}}
            
            if($BucketDescriptiveNames){
                if($bt -eq "terms"){
                    $this_bucket_name = $BucketField -replace "[^a-z0-9]","_"
                }else{
                    $this_bucket_name = $bt
                }
            }else{
                if($bucket_count -eq 1){
                    $this_bucket_name = "buckets"
                }else{
                    $this_bucket_name = "buckets$this_bucket_number"
                }

        $headers = @{Authorization=("Basic {0}" -f $base64AuthInfo)}
    }

    $fullURI = "$ELKrestURI/$ELKIndex/$documentType/$recordId"

    $body = @{doc=$record}

    if($pscmdlet.ParameterSetName -match "Hashtable"){
        $response = Invoke-RestMethod -Uri $fullURI -Method Post -Headers $headers -Body $($body.psobject.BaseObject | ConvertTo-Json -Depth 15) -ContentType 'application/json'
    }else{

    }

}

function Add-ElkShortUrl{
    #https://github.com/elastic/kibana/issues/12840
    #https://cinhtau.de/2022/04/11/kibana-shorturls/
    #https://observablehq.com/@nyurik/kibana-url-parser
    #https://discuss.elastic.co/t/kibana-g-and-a-parameters-in-the-dashboards-url-string/264642
    #https://www.elastic.co/guide/en/kibana/current/short-urls-api-create.html
    param(
        [CmdletBinding()]
        [Parameter(Mandatory=$false)] [string]$ELKrestURI = "http://localhost:5601",
        [Parameter(Mandatory=$false)] [string]$ELKSpace = "", 
        [Parameter(Mandatory=$false)] [PSCredential]$ElkCredential,
        [Parameter(Mandatory=$false)] [switch]$ElkCredentialIsAPIKey,
        [Parameter(Mandatory=$false)] [string]$ElkrestAPIendpoint='api/short_url',
        [Parameter(Mandatory=$false)] [ValidateSet("GET","PUT","POST","DELETE", "HEAD")] [string]$ElkRestMethod="POST",
        [Parameter(Mandatory=$false)] [string]$ElkApp='discover',
        [Parameter(Mandatory=$false)] [string]$HumanReadableSlug=((45..46)+(48..57)+(65..90)+(95)+(97..122) | Get-Random -count 42 | %{[char]$_}) -join '',
        [Parameter(Mandatory=$false)] [string[]]$ElkFiltersGlobal=@(),
        [Parameter(Mandatory=$false)] [string[]]$ElkFiltersApp=@(),
        [Parameter(Mandatory=$false)] [string]$ElkIndex='',
        [Parameter(Mandatory=$false)] [string[]]$ElkFields=@(),
        [Parameter(Mandatory=$false)] [string]$ElkTimeFrom='now-24h/h',
        [Parameter(Mandatory=$false)] [string]$ElkTimeTo='now',
        [Parameter(Mandatory=$false)] [bool]$ElkChartHide=$false,
        [Parameter(Mandatory=$false)] [string]$ElkChartInterval='auto',
        [Parameter(Mandatory=$false)] [int]$ElkRefreshIntervalSeconds=0,
        [Parameter(Mandatory=$false)] [bool]$ElkRefreshPause=$true,
        [Parameter(Mandatory=$false)] [string]$ElkQuery='',
        [Parameter(Mandatory=$false)] [hashtable]$ElkSortFields=@{'@timestamp'="desc"},
        [Parameter(Mandatory=$false,ParameterSetName="LongUrl")] [string]$ElkLongUrl='',
        [Parameter(Mandatory=$false)] [switch]$DebugRequest,
        [Parameter(Mandatory=$false)] [switch]$DebugQuery
    )
    if(!$ElkCredential -or $ELKcredential -eq $null -or $ELKcredential -eq "" -or $ELKcredential.GetType() -notmatch "credential"){
        $ElkCredential = Get-Credential -Message "Provide ELK Credential" -UserName $env:USERNAME
        $prompt = Read-Host -Prompt "Is this credential an API Key (y/N)?"
        if($prompt -match "^y"){$ElkCredentialIsAPIKey = $True}
    }

    # Convert creds to Base64
    $unsecureCreds = $ElkCredential.GetNetworkCredential()
    $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $unsecureCreds.UserName,$unsecureCreds.Password)))
    Remove-Variable unsecureCreds

    if($ElkCredentialIsAPIKey){
        $headers = @{'kbn-xsrf'=$true; 'Content-Type'= 'application/json'; Authorization=("ApiKey {0}" -f $base64AuthInfo)}
    }else{
        $headers = @{'kbn-xsrf'=$true; 'Content-Type'= 'application/json'; Authorization=("Basic {0}" -f $base64AuthInfo)}
    }

    If($ELKSpace){
        $ElkUrl = "$ELKrestURI$ELKSpace"
    }else{
        $ElkUrl = "$ELKrestURI"
    }

    $url_trailer = ""
    if($PSCmdlet.ParameterSetName -eq "LongUrl"){
        $url_trailer = $ElkLongUrl
    }else{
        $app_str = "app/$ElkApp#"
        
        $filters_global_str = "filters:!($($ElkFiltersGlobal -join ','))"
        
        if($ElkRefreshPause){$pause = "!t"}else{$pause="!f"}
        $refresh_str = "refreshInterval:(pause:$pause,value:$($ElkRefreshIntervalSeconds * 1000))"

        #$time_str = "time:(from:$([System.Web.HttpUtility]::UrlEncode($ElkTimeFrom)),to:$([System.Web.HttpUtility]::UrlEncode($ElkTimeTo)))"
        if($ElkTimeFrom -match "now"){
            $time_str1 = $ElkTimeFrom
        }else{
            $time_str1 = "'$ElkTimeFrom'"
        }
        if($ElkTimeTo -match "now"){
            $time_str2 = $ElkTimeTo
        }else{
            $time_str2 = "'$ElkTimeTo'"
        }
        $time_str = "time:(from:$time_str1,to:$time_str2)"
        
        for($i =0;$i -lt $ElkFields.Count;$i++){if($ElkFields[$i] -match "(\s)" -and $ElkFields[$i] -notmatch "^'.*'$"){$ElkFields[$i] = "'$($ElkFields[$i])'"}}

        $columns_str = "columns:!($(($ElkFields | % {[System.Web.HttpUtility]::UrlEncode($_)}) -join ','))"

        $filters_app_str = "filters:!($($ElkFiltersApp -join ','))"

        if($ElkChartHide){$hidechart_str = "hideChart:!t"}else{$hidechart_str = "hideChart:!f"}

        if($ElkIndex -match "[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{10}"){
            $index_str = "index:'$ElkIndex'"
        }else{
            $index_id = $(Get-ElkSavedObjectByName -search $ElkIndex -ObjectType index-pattern -ELKrestURI $ELKrestURI -ElkSpace $ELKSpace -ElkCredential $ElkCredential -ElkCredentialIsAPIKey:$ElkCredentialIsAPIKey | % saved_objects | where {$_.attributes.name -eq $ElkIndex} | select -First 1 | % id)
            if(!$index_id){
                $index_id = $(Get-ElkSavedObjectByName -search $ElkIndex -ObjectType index-pattern -ELKrestURI $ELKrestURI -ElkSpace $ELKSpace -ElkCredential $ElkCredential -ElkCredentialIsAPIKey:$ElkCredentialIsAPIKey | % saved_objects | where {$_.attributes.title -eq $ElkIndex} | select -First 1 | % id)
            }
            $index_str = "index:'$index_id'"
            if(!$index_id -or $index_id -eq ""){Write-Host -Debug -ForegroundColor Yellow "ERROR RESOLVING INDEX-PATTERN"; return}
        }

        $interval_str = "interval:$ElkChartInterval"

        $query_str = "query:(language:lucene,query:'$([System.Web.HttpUtility]::UrlEncode($ElkQuery))')"

        [string[]]$tmp_sort_str = [string[]]""
        [string[]]$tmp_sort_str = foreach ($field in $ElkSortFields.GetEnumerator()){
            if($field.key -match ""){
                "!('$($field.key)',$($field.value))"
            }else{
                "!($($field.key),$($field.value))"
            }
        }
        $sort_str = "sort:!(" +($tmp_sort_str -join ',') + ")"
        
        $url_trailer = "/$app_str/?_g=($filters_global_str,$refresh_str,$time_str)&_a=($columns_str,$filters_app_str,$hidechart_str,$index_str,$interval_str,$query_str,$sort_str)"

        $body = @{locatorId="LEGACY_SHORT_URL_LOCATOR";params=@{url=$url_trailer}}
        if($HumanReadableSlug -and $HumanReadableSlug -ne "" -and $HumanReadableSlug.Length -ge 3 -and $HumanReadableSlug.Length -le 255 -and $HumanReadableSlug -match "^[A-Za-z0-9\-\._]+$"){
            $body.Add("humanReadableSlug",$true)
            $body.Add("slug",$HumanReadableSlug)
        }else{
            Write-Host -Debug -ForegroundColor Yellow "Invalid slug: $HumanReadableSlug"
        }

        #The data in the URL parameters is the application state that has been encoded in "Rison" format. The _a parameter is "app state" and the _g parameter is "global state", which is state that moves between apps with you (such as pinned filters).
        #RISON Encoding  https://github.com/Nanonid/rison
        #You can use a tool such as the following to make sense of the data: https://observablehq.com/@nyurik/kibana-url-parser
        #$url_trailer = $ElkUrl+"/$app_str/?_g="
    }
    if($DebugQuery){
        Write-Host -Debug -ForegroundColor Yellow $ElkUrl
        Write-Host -Debug -ForegroundColor Yellow $($body | ConvertTo-Json -Depth 10 | Out-String)
    }
    if($DebugRequest){
        Write-Host -Debug -ForegroundColor Yellow "$ElkrestMethod $ElkUrl/$ElkrestAPIendpoint"
    }
    $response = Invoke-RestMethod "$ElkUrl/$ElkrestAPIendpoint" -Method $ElkrestMethod -Headers $headers -ContentType 'application/json' -Body $($body | ConvertTo-Json -Depth 10)
    $response | Add-Member -MemberType NoteProperty -Name "short_url" -Value ("$ElkUrl/app/r/s/$($response.slug)")
    $response | Add-Member -MemberType NoteProperty -Name "goto_url" -Value ("$ElkUrl/goto/$($response.id)")
    $response | Add-Member -MemberType NoteProperty -Name "long_url" -Value ("$ElkUrl/$url_trailer")
    return $response
}

        [Parameter(Mandatory=$false)] $ELKRestEndpoint = "api/short_url",
        [Parameter(Mandatory=$false)] [PSCredential]$ElkCredential,
        [Parameter(Mandatory=$false)] [switch]$ElkCredentialIsAPIKey
    )
    if(!$ElkCredential -or $ELKcredential -eq $null -or $ELKcredential -eq "" -or $ELKcredential.GetType() -notmatch "credential"){
        $ElkCredential = Get-Credential -Message "Provide ELK Credential" -UserName $env:USERNAME
        $prompt = Read-Host -Prompt "Is this credential an API Key (y/N)?"
        if($prompt -match "^y"){$ElkCredentialIsAPIKey = $True}
    }

    # Convert creds to Base64
    $unsecureCreds = $ElkCredential.GetNetworkCredential()
    $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $unsecureCreds.UserName,$unsecureCreds.Password)))
    Remove-Variable unsecureCreds

    if($ElkCredentialIsAPIKey){
        $headers = @{'kbn-xsrf'=$true; 'Content-Type'= 'application/json'; Authorization=("ApiKey {0}" -f $base64AuthInfo)}
    }else{
        $headers = @{'kbn-xsrf'=$true; 'Content-Type'= 'application/json'; Authorization=("Basic {0}" -f $base64AuthInfo)}
    }

    if($slug){
        $ElkUrl = "$ELKrestURI/$ElkSpace/$ELKRestEndpoint/_slug/$slug"
    }else{
        $ElkUrl = "$ELKrestURI/$ElkSpace/$ELKRestEndpoint/$guid"
    }

    $response = Invoke-RestMethod $ElkUrl -Method Get -Headers $headers -ContentType 'application/json' #-UseBasicParsing

    if(!$response){return}

    $trycount = 0
    while(($response.GetType() -match "string") -and ($trycount -lt 4)){
        $term = ''
        try{
            $response = $response | ConvertFrom-Json
        }catch{
            $errormessage = $_.Exception.Message 
            if($errormessage -match "duplicated keys"){
                if($verbose){Write-Host -ForegroundColor Yellow -Debug $errormessage}
                $term = $errormessage -replace "'","" -replace "\.",'' -split ' ' | select -Last 1
                if($verbose){Write-Host -ForegroundColor Yellow -Debug "replacing term: '$term'"}
                if($term){$response = $response -replace $term.ToLower(),"$($term)_2"}
                
            }
            $trycount++
        }
    }
    return $response
}

function Get-DayLightSavingTime {
 
<#
.SYNOPSIS
    Returns the beginning and ending date for daylight saving time.
.DESCRIPTION
    Get-DayLightSavingTime is a function that returns the dates when
    daylight saving time begins and ends for the specified year.
.PARAMETER Year
    The year to return the daylight saving time dates for. The year cannot
    be earlier than 2007 because the dates were in April and October instead
    of March and November prior to that year. The default is the current year.
.EXAMPLE
    Get-DayLightSavingTime
.EXAMPLE
    Get-DayLightSavingTime -Year 2014, 2015
.EXAMPLE
    Get-DayLightSavingTime -Year (2011..2020)
.EXAMPLE
    2014, 2015 | Get-DayLightSavingTime
.EXAMPLE
    2011..2020 | Get-DayLightSavingTime
.INPUTS
    Integer
.OUTPUTS
    PSCustomObject
.NOTES
    Written by Mike F Robbins
    Blog: http://mikefrobbins.com
    Twitter: @mikefrobbins
#>
 
    [CmdletBinding()]
    param (
        [Parameter(ValueFromPipeline)]
        [ValidateRange(2007,9999)]
        [Int[]]$Year = (Get-Date).Year
    )
 
    PROCESS {
        foreach ($y in $Year) {
 
            [datetime]$beginDate = "March 1, $y"
 
            while ($beginDate.DayOfWeek -ne 'Sunday') {
                $beginDate = $beginDate.AddDays(1)
            }
 
            [datetime]$endDate = "November 1, $y"
 
            while ($endDate.DayOfWeek -ne 'Sunday') {
                $endDate = $endDate.AddDays(1)
            }            
 
            [PSCustomObject]@{
                'Year' = $y
                'BeginDate' = $($beginDate.AddDays(7).AddHours(2))
                'EndDate' = $($endDate.AddHours(2))
            }
 
        }
    }
}

function Get-RecursiveComparison{
    param(
        [hashtable]$obj1,
        [hashtable]$obj2,
        [string]$prefix="",
        [switch]$IncludeEqual
    )
    if($prefix){$prefix += '.'}
    
    foreach ($item in $obj1.GetEnumerator()){
        if($item.Value.GetType().Name -match "hashtable" -and $obj2 -and $obj2.$($item.Name) -and $obj2.$($item.Name).GetType().Name -match "hashtable"){
            Get-RecursiveComparison -obj1 $item.Value -obj2 $obj2.$($item.Name)  -prefix "$prefix$($item.Name)"
        }elseif($item.Value.GetType().Name -match "hashtable" -and (!$obj2 -or $obj2.Count -eq 0 -or !$obj2.$($item.Name) -or !$obj2.ContainsKey($item.Name) -or !($obj2.$($item.Name).GetType().Name -match "hashtable"))){
            $r = "" | select Field,Object1,Operator,Object2
            $r.Field = "$prefix$($item.Name)"
            $r.Object1 = Get-FlattenedHashtable -hashtable $item.Value -prefix $prefix
            if($obj2.ContainsKey($item.Name)){
                $r.Object2 = $obj2.$($item.Name)
            }else{
                $r.Object2 = ""
            }
            $r.Operator = "<="
            $r
        }else{
            if($obj2.Count -eq 0 -or !$obj2.ContainsKey($item.Name)){
                $r = "" | select Field,Object1,Operator,Object2
                $r.Field = "$prefix$($item.Name)"
                if($item.Value.GetType().Name -match "hashtable"){
                    $r.Object1 = Get-FlattenedHashtable -hashtable $item.Value -prefix $prefix
                }else{
                    $r.Object1 = $item.Value
                }
                $r.Object2 = " "
                $r.Operator = "<="
                $r    
            }else{
                $c = Compare-Object $item.Value $obj2.$($item.Name)
            
                if(!$c -and $IncludeEqual){
                    $r = "" | select Field,Object1,Operator,Object2
                    $r.Field = "$prefix$($item.Name)"
                    $r.Object1 = $item.Value
                    $r.Object2 = $obj2.$($item.Name)
                    $r.Operator = "="
                    $r
                }elseif($c){
                    $r = "" | select Field,Object1,Operator,Object2
                    $r.Field = "$prefix$($item.Name)"
                    $r.Object1 = $item.Value
                    $r.Object2 = $obj2.$($item.Name)
                    $r.Operator = $c.SideIndicator
                    $r
                }
            }
        }
    }
    foreach ($item in $obj2.GetEnumerator()){
        if($item.Value.GetType().Name -match "hashtable" -and $obj1 -and $obj1.$($item.Name) -and $obj1.$($item.Name).GetType().Name -match "hashtable"){
            #do nothing - we already handled this condition above
        }elseif($item.Value.GetType().Name -match "hashtable" -and (!$obj1 -or $obj1.Count -eq 0 -or !$obj1.$($item.Name) -or !$obj1.ContainsKey($item.Name) -or !($obj1.$($item.Name).GetType().Name -match "hashtable"))){
            $r = "" | select Field,Object1,Operator,Object2
            $r.Field = "$prefix$($item.Name)"
            if($obj1.ContainsKey($item.Name)){
                $r.Object1 = $obj1.$($item.Name)
            }else{
                $r.Object1 = ""
            }
            $r.Object2 = Get-FlattenedHashtable -hashtable $item.Value -prefix $prefix
            $r.Operator = "=>"
            $r
        }else{
            if($obj1.Count -eq 0 -or !$obj1.ContainsKey($item.Name)){
                $r = "" | select Field,Object1,Operator,Object2
                $r.Field = "$prefix$($item.Name)"
                $r.Object1 = " "
                if($item.Value.GetType().Name -match "hashtable"){
                    $r.Object2 = Get-FlattenedHashtable -hashtable $item.Value -prefix $prefix
                }else{
                    $r.Object2 = $item.Value
                }
                $r.Operator = "=>"
                $r    
            }else{
                $c = Compare-Object $obj1.$($item.Name) $item.Value 
            
                if(!$c -and $IncludeEqual){
                    #Do Nothing we should have already handled this when looking at Obj1 above
                }elseif($c){
                    $r = "" | select Field,Object1,Operator,Object2
                    $r.Field = "$prefix$($item.Name)"
                    $r.Object1 = $obj2.$($item.Name)
                    $r.Object2 = $item.Value
                    $r.Operator = $c.SideIndicator
                    $r
                }
            }
        }
    }
}

function Get-FlattenedHashtable{
    param(
        [hashtable]$hashtable,
        [string]$prefix=""
    )
    if($prefix){$prefix += '.'}
    if(!$hashtable){
        $null
    }else{
        foreach($item in $hashtable.GetEnumerator()){
            if($item.Value.GetType().Name -match "hashtable"){
            
                Get-FlattenedHashtable -hashtable $item.Value -prefix "$prefix$($item.Name)"
            }else{
                $o = "" | select Name,Value
                $o.Name = "$prefix$($item.Name)"
                $o.Value = $item.Value
                $o
            }
        }
    }
}

function Get-ProcessTree{
    param(
        [CmdletBinding()]
        [Parameter(Mandatory=$false)] $ELKrestURI = "http://localhost:9200",
        [Parameter(Mandatory=$false)] [PSCredential]$ElkCredential,
        [Parameter(Mandatory=$false)] [switch]$ElkCredentialIsAPIKey,
        [Parameter(Mandatory=$false)] [string]$ELKIndex="hunt*",
        [Parameter(Mandatory=$true)]  [string]$seedGUID,
        [Parameter(Mandatory=$false)] [switch]$allParents,
        [Parameter(Mandatory=$false)] [switch]$noParents,
        [Parameter(Mandatory=$false)] [int]$MaxGUIDs=400,
        [Parameter(Mandatory=$false)] [string]$ParentImageCutoffRegex="\\(services|explorer|winlogon|userinit|svchost).exe",
        [Parameter(Mandatory=$false, ParameterSetName="Fields")] [hashtable]$EventIdDisplayFields=@{1=@("CommandLine");8=@("SourceImage","TargetImage");11=@("TargetFilename");22=@("QueryName","QueryResults")},
        [Parameter(Mandatory=$false, ParameterSetName="Fields")] [switch]$NoEventDescriptions,
        [Parameter(Mandatory=$false, ParameterSetName="Fields")] [switch]$NoTimestamps,
        [Parameter(Mandatory=$false)] [ValidateRange(1,6)] [int]$Verbosity=6,
        [Parameter(Mandatory=$false)] [switch]$JustShowMeTheTree,        
        [Parameter(Mandatory=$false)] [ValidateSet("Chronological", "PreserveTree")] [string]$SortOrder="PreserveTree",
        [Parameter(Mandatory=$false)] [string]$CSVOutputFile,
        [Parameter(Mandatory=$false)] [int]$numDaysAgoStart=30,
        [Parameter(Mandatory=$false)] [int]$numDaysAgoEnd=0        
    )

    Get-TreeArt

    if(!($EventIdDisplayFields.Keys -contains "1")){
        $EventIdDisplayFields.Add(1,@("CommandLine"))
        Write-Warning "Adding event ID 1 since that is necessary for process chanining"
    }

    switch ($Verbosity){
        1 { $EventIdDisplayFields=@{1=@("CommandLine")} }
        2 { $EventIdDisplayFields=@{1=@("User","CommandLine");8=@("SourceImage","TargetImage");22=@("QueryName","QueryResults")} }
        3 { $EventIdDisplayFields=@{1=@("User","CommandLine");8=@("SourceImage","TargetImage");11=@("TargetFilename");22=@("QueryName","QueryResults")} }
        4 { $EventIdDisplayFields=@{1=@("User","OriginalFileName","CommandLine");3=@("SourceIp","SourcePort";"DestinationHostname","DestinationIp","DestinationPort")
            8=@("SourceImage","TargetImage");11=@("TargetFilename");12=@("TargetObject");13=@("TargetObject","Details");14=@("TargetObject");22=@("QueryName","QueryResults")} }
        5 { $EventIdDisplayFields=@{1=@("host","HostSourceAddress","User","OriginalFileName","Image","CurrentDirectory","CommandLine");
            3=@("SourceIp","SourcePort";"DestinationHostname","DestinationIp","DestinationPort","Protocol","Initiated")
            8=@("SourceImage","TargetImage");11=@("TargetFilename");12=@("TargetObject");13=@("TargetObject","Details");14=@("TargetObject");22=@("QueryName","QueryResults","QueryStatus")} }
        6 { $EventIdDisplayFields=@{1=@("host","HostSourceAddress","User","OriginalFileName","Image","CurrentDirectory","CommandLine","Hashes");
            3=@("SourceIp","SourcePort";"DestinationHostname","DestinationIp","DestinationPort","Protocol","Initiated")
            8=@("SourceImage","TargetImage");11=@("TargetFilename");12=@("TargetObject");13=@("TargetObject","Details");14=@("TargetObject");22=@("QueryName","QueryResults","QueryStatus")} }
    }
    
    $GUIDlist = New-Object -TypeName System.Collections.ArrayList
    $pendingGUIDs = New-Object -TypeName System.Collections.ArrayList
    [void]$pendingGUIDs.Add($seedGuid)
    $global:GUIDtree = @{$seedGUID= @{}}
    $result = $null
    $GUIDlist_str = ''
    $pass = 1
    
    Write-Host -Debug -ForegroundColor Green "Searching for related GUIDs in Process Tree..."
    while($pendingGUIDs.Count -gt 0 -and $GUIDlist.Count -le $MaxGUIDs){
        $pendingGUIDs | %{[void]$GUIDlist.Add($_)}
        $pendingGUIDs.Clear()
        
        $GUIDlist_str = ''
        $GUIDlist | %{$GUIDlist_str+='"'+$_+'" OR '}
        $GUIDlist_str = $GUIDlist_str.Substring(0, $GUIDlist_str.Length -4)

        $result = Get-ELKRecords -ELKrestURI $ELKrestURI -ElkCredential $ElkCredential -ElkCredentialIsAPIKey:$ElkCredentialIsAPIKey -ELKIndex $ElkIndex -ELKQuery $('event_id:1 AND (event_data.ProcessGuid:('+$GUIDlist_str+') OR event_data.ParentProcessGuid:('+$GUIDlist_str+'))') -numDaysAgoStart $numDaysAgoStart -numDaysAgoEnd $numDaysAgoEnd
        
        if(!$JustShowMeTheTree){
            Write-Host -Debug -ForegroundColor Green "Pass #$($pass): Hit Count:$($result.hits.total.value)"
        }

        if($result.hits.total.value -gt $MaxGUIDs -or $GUIDlist.Count -gt $MaxGUIDs){
            Write-Host -Debug -ForegroundColor Yellow "MaxGUIDs value exceeded. Invoking Lumberjack protocol!"
            break
        }
        
        if($result -and $result.hits.total.value -gt 0 -and $result.hits.total.value -le $MaxGUIDs){
            foreach($hit in $result.hits.hits){
                $global:GUIDtree = [hashtable](Insert-RecursiveGUID -ParentGUID $hit._source.event_data.ParentProcessGuid -ChildGUID $hit._source.event_data.ProcessGuid -Tree $global:GUIDtree)
                #Print-GUIDTree $GUIDtree
                if(!$noParents -and !($GUIDlist.Contains($hit._source.event_data.ParentProcessGuid)) -and !($pendingGUIDs.Contains($hit._source.event_data.ParentProcessGuid)) -and ($allParents -or ($hit._source.event_data.ParentImage -notmatch $ParentImageCutoffRegex) ) ){
                    [void]$pendingGUIDs.Add($hit._source.event_data.ParentProcessGuid)
                }
                if(!($GUIDlist.Contains($hit._source.event_data.ProcessGuid)) -and !($pendingGUIDs.Contains($hit._source.event_data.ProcessGuid))){
                    [void]$pendingGUIDs.Add($hit._source.event_data.ProcessGuid)
                }
            }
        }
        $pass++
    }
    
    $eventIdStr = ""
    if($EventIdDisplayFields.Keys -match "[2-9]{1,2}|[0-9]{2,3}"){
        $eventIdStr = "("
        foreach($id in $EventIdDisplayFields.GetEnumerator()){
            $eventIdStr += "$($id.key)" + ' OR '
        }
        $eventIdStr = $eventIdStr.SubString(0,$($eventIdStr.length-4)) + ') '
        $result = Get-ELKRecords -ELKrestURI $ELKrestURI -ElkCredential $ElkCredential -ELKIndex "*:ls-wlbsysmon" -ELKQuery $('event_id:'+$eventIdStr+' AND (event_data.ProcessGuid:('+$GUIDlist_str+') OR event_data.ParentProcessGuid:('+$GUIDlist_str+'))') -numDaysAgoStart $numDaysAgoStart -numDaysAgoEnd 0
    }else{
        $eventIdStr = "1"
    }

    if(!$JustShowMeTheTree){
        Write-Host -Debug -ForegroundColor Magenta "----------FINAL QUERY-----------";
        Write-Host -Debug -ForegroundColor Green $('event_id:'+$eventIdStr+' AND (event_data.ProcessGuid:('+$GUIDlist_str+') OR event_data.ParentProcessGuid:('+$GUIDlist_str+'))');
        Write-Host -Debug -ForegroundColor Magenta "-----------END----------`n"
    }

    $parsedResults = New-Object -TypeName System.Collections.ArrayList
    foreach($hit in $result.hits.hits){
        $record = $hit._source | Select-Object eventTimestamp,event_id,computer_name,host,ship_time,index_time
        $tmp2 = $hit._source | Select-Object -ExpandProperty event_data
        $tmp2.psobject.properties| %{Add-member -InputObject $record -NotePropertyName $_.Name -NotePropertyValue $_.Value}
        Add-Member -InputObject $record -NotePropertyName "_id" -NotePropertyValue $($hit | Select-Object -ExpandProperty _id)
        if($hit._source.SourceAddress){ 
            Add-Member -InputObject $record -NotePropertyName "HostSourceAddress" -NotePropertyValue $($hit._source | Select-Object -ExpandProperty SourceAddress)
        }else{
            Add-Member -InputObject $record -NotePropertyName "HostSourceAddress" -NotePropertyValue "HostSourceAddress Not Available"
        }

        [void]$parsedResults.Add($record)
    }

    $root = $GUIDtree.GetEnumerator() | select -First 1
    $answers = New-Object -TypeName System.Collections.ArrayList
    if($NoTimeStamps){
        $result.hits.hits._source.event_data | where ParentProcessGuid -match $root.Key | %{$str =  "$prefix$($_.ParentCommandLine)"}
        [void]$answers.Add($str)
    }else{
        $rootRecord = Get-ELKRecords -ELKrestURI $ELKrestURI -ElkCredential $ElkCredential -ELKIndex "*:ls-wlbsysmon" -ELKQuery $('event_id:1 AND event_data.ProcessGuid:"'+$root.key+'"') -numDaysAgoStart $($numDaysAgoStart+7) -numDaysAgoEnd $numDaysAgoEnd
        if($rootRecord -and $rootRecord.hits.total.value -eq 1){
            $str = "$($rootRecord.hits.hits._source.eventTimestamp) $($rootRecord.hits.hits._source.event_data.ParentImage)->$($rootRecord.hits.hits._source.event_data.CommandLine)"
            [void]$answers.Add($str)
        }else{
            $str = $result.hits.hits._source.event_data | where ParentProcessGuid -match $root.Key | select -First 1 | Select-Object -ExpandProperty ParentCommandLine | %{"$prefix$($_)"}
            if($str -eq $null){
                Write-Host -ForegroundColor Red -Debug "Unable to locate root node for that seedGuid. Try expanding your time window in case the process was created several days/weeks before the event/seedGuid occurred."
                Write-Host -ForegroundColor Red -Debug "Attempting to print records anyway..."
                $parsedResults | select-object -expandProperty ProcessGuid | sort-object | get-unique | %{ if (!$global:GUIDTree.ContainsKey($_)){ $global:GUIDtree.add($_, @{})}}
            }else{
                [void]$answers.Add("$prefix$($str)")
            }
        }        
    }
    If($answers.Count -eq 0){[Void]$answers.Add("Unable to Locate Root/Parent node")}
    
    if($parsedResults.Count -eq 0){
        Write-Host -ForegroundColor Red -Debug "No records were returned for that final query. Try expanding your time window with the '-numDaysAgoStart' parameter, it is likely that the seedGUid you observed was logged more than $numDaysAgoStart days ago"
        return
    }
    $answers = Print-ProcessTree -Tree $GUIDtree -records $parsedResults -EventIdDisplayFields $EventIdDisplayFields -NoTimestamps $NoTimestamps.IsPresent -NoEventDescription $NoEventDescriptions.IsPresent -answer $answers.clone()

    if($SortOrder -eq "Chronological"){
        $answers | sort | Write-Host -Debug -ForegroundColor Green
    }else{
        $answers | Write-Host -Debug -ForegroundColor Green
    }
    return $result
}

function Insert-RecursiveGUID{
    param(
        [Parameter(Mandatory=$true)]
        [string]$ParentGUID,
        [Parameter(Mandatory=$true)]
        [string]$ChildGUID,
        [Parameter(Mandatory=$true)]
        [hashtable]$Tree
    )
    if(($Tree.Count -gt 0) -and ($Tree.GetType().Name -match "hashtable") -and $Tree.ContainsKey($ParentGUID) -and !$Tree.$ParentGUID.ContainsKey($ChildGUID)){
        ([hashtable]($Tree.$ParentGUID)).Add($ChildGUID, @{})
        return $Tree
    }elseif(($Tree.Count -gt 0) -and $Tree.ContainsKey($ParentGUID) -and $Tree.$ParentGUID.ContainsKey($ChildGUID)){
        return $Tree
    }elseif(($Tree.Count -gt 0) -and $Tree.ContainsKey($ChildGUID)){
        return @{$ParentGUID=$Tree}
    }elseif(($Tree.Count -gt 0) -and ($Tree.Values -ne $null)){
        $newTree = $Tree.Clone()
        $newValues = @{}
        foreach($k in $Tree.keys){
            if($Tree.$k -eq $null){continue}
            $newValues = (Insert-RecursiveGUID -ParentGUID $ParentGUID -ChildGUID $ChildGUID -Tree $Tree.$k)
            $NewTree.$k = $newValues
        }
        return $NewTree
    }elseif($tree.Count -eq 0){
        return @{}
    }else{
        return @{}
    }
}

        [Parameter(Mandatory=$true)]
        [hashtable]$Tree,
        [Parameter(Mandatory=$false)]
        $prefix=""
    )

        if($item -ne $null -and $item -ne "" -and $item.key -ne $null -and $item.key -ne ""){write-host "$($prefix)$([string]$item.key)"}

    }
}

function Print-ProcessTree{
    param(
        [Parameter(Mandatory=$true)]
        [hashtable]$Tree,
        [Parameter(Mandatory=$true)]
        [Object[]]$records,
        [Parameter(Mandatory=$false)]
        [string]$prefix="",
        [Parameter(Mandatory=$false)]
        [bool]$NoTimestamps,
        [Parameter(Mandatory=$false)]
        [bool]$NoEventDescription=$false,
        [Parameter(Mandatory=$false)]
        [hashtable]$EventIdDisplayFields,
        [Parameter(Mandatory=$false)]
        [string]$FieldDelimiter=" | ", # ",","`t"
        [Parameter(Mandatory=$true)]
        [System.Collections.ArrayList]$answer
    )
    $eventDescriptions = @{1="Process Created";3="NetTraffic";8="ProcessInjected";11="FileCreated";12="RegKeyCreated/Deleted";13="RegKeyValueSet";14="RegKeyRenamed";15="StreamCreated";17="PipeCreated";
        18="PipeConnected";22="DNSQuery";25="ProcessTampered"}

    if(!$tree -or ($tree -eq $null) -or ($tree.count -eq 0)){return $answer}

    foreach($item in $Tree.GetEnumerator()){
        if($item -ne $null -and $item -ne "" -and $item.key -ne $null -and $item.key -ne ""){
            
            $GuidEvents = $records | where ProcessGuid -match $item.key
            foreach($EIDF in ($EventIdDisplayFields.GetEnumerator() | Sort-Object -Property Key)){
                $DisplayFields = $EIDF.Value
                $id = $EIDF.Key
                
                $GuidIdEvents = $GuidEvents | where event_id -eq $id
                $str = ""
                $desc = ""
                if($NoEventDescription){
                    $desc = "event_id:$id"
                }else{
                    if($eventDescriptions.ContainsKey($id)){
                        $desc = $eventDescriptions.$id
                    }else{
                        $desc = "event_id:$id"
                    }                    
                }
                foreach($e in $GuidIdEvents){
                    $thisStr = ""
                    if($NoTimestamps){
                        $thisStr += "$prefix$desc"
                    }else{
                        $thisStr = "$($e.eventTimestamp.ToString()) $prefix$desc$FieldDelimiter"
                    }
                    foreach($field in $DisplayFields){
                        $thisStr += $e | Select-Object -ExpandProperty $field
                        $thisStr += $FieldDelimiter
                    }
                    $thisStr = $thisStr.Substring(0,$thisStr.Length - 3)
                    if($thisStr -and ($thisStr.Trim() -ne "")){[void]$answer.Add($thisStr);}
                }

            }
        }
        $answer = Print-ProcessTree -Tree $item.value -records $records -EventIdDisplayFields $EventIdDisplayFields -prefix $($prefix+"-") -NoTimestamps $NoTimestamps -NoEventDescription $NoEventDescription -answer $answer.Clone()
        
    }
    return $answer
}

function ConvertTo-Unicode{
    param([string]$char)

    #Regex that can be used to identify strings that have unicode characters where this function might be necessary
    #$nonASCII = $str -match "[^\x00-\x7F]"
    #sample usage
    <#
        $jsonquery | sls "[^\x00-\x7F]" -AllMatches | % Matches | %{
            $jsonquery_unicode = $jsonquery_unicode -replace $_.Value,$(ConvertTo-Unicode -char $_.Value)
        }
    #>

    $enc = [System.Text.Encoding]::UTF32.GetBytes($char)
    $str = [System.String]::Format("{0:X}",($enc | select -First 1))
    if($str.length -eq 1){$str += "0"}
    return "\u00$str"
}

function Get-FieldCount($obj){
    if($obj.GetType().name -notmatch "PSCustomObject"){
        return 1
    }else{
        $count = 0;
        foreach($subobj in ($obj.psobject.properties | ? MemberType -eq NoteProperty | ? TypeNameOfValue -Match "PSCustomObject")){
            $count+=count_fields -obj $subobj.Value
        };
        foreach($field in ($obj.psobject.properties | ? MemberType -eq NoteProperty | ? TypeNameOfValue -NotMatch "PSCustomObject")){
            $count++
        };
        return $count
    } 
}
